\subsection*{الف}
بازهٔ کوانتیزه‌سازی $[-3.5,3.5]$ طولش $7$ است و چون کوانتیزر ۲بیتی داریم، تعداد ترازهای بازسازی $L=4$ خواهد بود. پس گامِ یکنواخت:
\[
\Delta=\frac{7}{4}=1.75.
\]
سطوح تصمیم (با درنظر گرفتن کران‌های ابتدا و انتها):
\[
b_0=-\frac{7}{2},\quad b_1=-\frac{7}{4},\quad b_2=0,\quad b_3=\frac{7}{4},\quad b_4=\frac{7}{2}.
\]
سطوح بازسازی (میانگین هر بازه):
\[
r_1=\frac{b_0+b_1}{2}=-\frac{21}{8},\quad
r_2=\frac{b_1+b_2}{2}=-\frac{7}{8},\quad
r_3=\frac{b_2+b_3}{2}=\frac{7}{8},\quad
r_4=\frac{b_3+b_4}{2}=\frac{21}{8}.
\]
قاعدهٔ انتساب (با قرارداد صورت سؤال: روی آستانه، بازهٔ سمت راست انتخاب می‌شود):
\begin{flalign*}
Q(x)=
\begin{cases}
r_1 & -\frac{7}{2}\le x<-\frac{7}{4}\\
r_2 & -\frac{7}{4}\le x<0\\
r_3 & 0\le x<\frac{7}{4}\\
r_4 & \frac{7}{4}\le x\le \frac{7}{2}
\end{cases} &&
\end{flalign*}

\subsection*{ب}
برای مقادیر منبع
$x\in\{-3,-2,-1,0,1,2,3\}$
داریم:
\[
Q(-3)=Q(-2)=-\frac{21}{8},\quad
Q(-1)=-\frac{7}{8},\quad
Q(0)=Q(1)=\frac{7}{8},\quad
Q(2)=Q(3)=\frac{21}{8}.
\]
خطای میانگین مربعی:
\[
\mathrm{MSE}=\mathbb{E}\big[(x-Q(x))^2\big]=\sum_x P(x)\,(x-Q(x))^2.
\]
\begin{flalign*}
\mathrm{MSE}
&=0.05\left(-3+\frac{21}{8}\right)^2
+0.10\left(-2+\frac{21}{8}\right)^2
+0.15\left(-1+\frac{7}{8}\right)^2
+0.40\left(0-\frac{7}{8}\right)^2 \\
&\quad +0.15\left(1-\frac{7}{8}\right)^2
+0.10\left(2-\frac{21}{8}\right)^2
+0.05\left(3-\frac{21}{8}\right)^2 \\
&=0.05\left(\frac{-3}{8}\right)^2
+0.10\left(\frac{5}{8}\right)^2
+0.15\left(\frac{-1}{8}\right)^2
+0.40\left(\frac{-7}{8}\right)^2
+0.15\left(\frac{1}{8}\right)^2
+0.10\left(\frac{-5}{8}\right)^2
+0.05\left(\frac{3}{8}\right)^2 \\
&=\frac{129}{320}\approx 0.403125. &&
\end{flalign*}

\subsection*{ج}
در یک گام \lr{Lloyd-Max} (با همان نواحی تصمیمِ قسمت الف)، ابتدا سطح بازسازی هر ناحیه را به‌صورت مرکزِ جرم (میانگین شرطی) حساب می‌کنیم.

\[
\mathcal{R}_1=\{-3,-2\},\;
\mathcal{R}_2=\{-1\},\;
\mathcal{R}_3=\{0,1\},\;
\mathcal{R}_4=\{2,3\}.
\]

\begin{flalign*}
r_1^{(new)}&=\frac{-3(0.05)+-2(0.10)}{0.05+0.10}=\frac{-0.35}{0.15}=-\frac{7}{3}, \\
r_2^{(new)}&=\frac{-1(0.15)}{0.15}=-1, \\
r_3^{(new)}&=\frac{0(0.40)+1(0.15)}{0.40+0.15}=\frac{0.15}{0.55}=\frac{3}{11}, \\
r_4^{(new)}&=\frac{2(0.10)+3(0.05)}{0.10+0.05}=\frac{0.35}{0.15}=\frac{7}{3}. &&
\end{flalign*}

سپس سطوح تصمیمِ جدید را با نیمساز بین سطوح بازسازی به‌دست می‌آوریم:
\begin{flalign*}
b_1^{(new)}&=\frac{r_1^{(new)}+r_2^{(new)}}{2}=\frac{-\frac{7}{3}+(-1)}{2}=-\frac{5}{3},\\
b_2^{(new)}&=\frac{r_2^{(new)}+r_3^{(new)}}{2}=\frac{-1+\frac{3}{11}}{2}=-\frac{4}{11},\\
b_3^{(new)}&=\frac{r_3^{(new)}+r_4^{(new)}}{2}=\frac{\frac{3}{11}+\frac{7}{3}}{2}=\frac{43}{33}. &&
\end{flalign*}
(کران‌های ابتدا/انتها همان $b_0=-3.5$ و $b_4=3.5$ در نظر گرفته می‌شوند.)

\subsection*{د}
در معیار \lr{MSE}:
\begin{itemize}
    \item اگر مرزها ثابت باشند، بهترین سطح بازسازی هر ناحیه همان مرکزِ جرم آن ناحیه است (گامِ \lr{centroid update}) و اعوجاج را کمینه می‌کند.
    \item اگر سطوح بازسازی ثابت باشند، بهترین مرز تصمیم بین دو سطح مجاور، نیمساز آن‌هاست (گامِ \lr{midpoint update}).
\end{itemize}
پس تکرار این دو گام نوعی کمینه‌سازی تکراری است که معمولاً اعوجاج را در هر تکرار کاهش می‌دهد و به یک جواب پایدار (کمینهٔ موضعی) می‌رسد.

\subsection*{هـ}
تقریب
\[
\mathrm{SNR}\approx 6B~\mathrm{dB}
\]
وقتی معنی‌دار است که:
\begin{itemize}
    \item کوانتیزر یکنواخت $B$ بیتی در ناحیهٔ \lr{high-resolution} کار کند (گام نسبت به تغییرات سیگنال کوچک باشد).
    \item خطای کوانتیزه‌سازی را بتوان نویز یکنواخت و تقریباً مستقل از سیگنال فرض کرد (مدل نویز سفید کوانتیزه‌سازی).
    \item \lr{overload} رخ ندهد و سیگنال از کل بازهٔ دینامیکی به‌صورت مناسب استفاده کند (مثلاً برای سینوسِ تمام‌مقیاس، رابطهٔ دقیق‌تر $\mathrm{SNR}\approx 6.02B+1.76$ dB است).
\end{itemize}

\subsection*{و}
\lr{Compandor} سه بخش دارد:
\[
x \xrightarrow{\text{compressor }\,c(\cdot)} y \xrightarrow{\text{uniform quantizer}} \hat y \xrightarrow{\text{expander }\,c^{-1}(\cdot)} \hat x.
\]
چون تابع فشرده‌ساز $c(\cdot)$ معمولاً برای دامنه‌های کوچک شیب بزرگ‌تری دارد، بعد از کوانتیزه‌سازی یکنواخت در دامنهٔ $y$، در دامنهٔ اصلی $x$ عملاً گام‌های ریزتر برای مقادیر کوچک و گام‌های درشت‌تر برای مقادیر بزرگ به‌دست می‌آید؛ یعنی رفتار یک کوانتیزر غیر یکنواخت (شبیه \lr{nonuniform quantizer}) بدون تغییرِ سخت‌افزارِ کوانتیزر یکنواخت.
